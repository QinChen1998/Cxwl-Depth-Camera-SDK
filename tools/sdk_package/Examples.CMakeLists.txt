cmake_minimum_required(VERSION 3.21)
project(CXWLDepthExamples LANGUAGES C CXX RC)
find_package(CxwlDepthSdk 3.0 CONFIG REQUIRED PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../SDK/cmake" NO_DEFAULT_PATH)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
option(CXWL_BUILD_QT_VIEWER "Build the Qt Widgets viewer" OFF)
function(cxwl_example target source profile)
  add_executable(${target} "${source}")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/common")
  target_link_libraries(${target} PRIVATE CXWL::DepthSdk)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc"
    CONTENT "#define CXWL_BINARY_NAME \"${target}.exe\"\n#define CXWL_BINARY_TYPE VFT_APP\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/common/ProductVersion.rc\"\n")
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}-version.rc")
  # Separate output directories keep the offline example free of capture dependencies.
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${profile}")
  cxwl_deploy_runtime(${target} PROFILE ${profile})
endfunction()
cxwl_example(CxwlDepthQuickStart "c/QuickStart/QuickStart.c" STANDARD)
cxwl_example(CxwlDepthQuality "c/DepthQuality/DepthQuality.c" STANDARD)
cxwl_example(CxwlDepthDeviceManagement "c/DeviceManagement/DeviceManagement.c" STANDARD)
cxwl_example(CxwlDepthPointCloud "c/PointCloud/PointCloud.c" STANDARD)
cxwl_example(CxwlDepthCppQuickStart "cpp/QuickStart/CppQuickStart.cpp" STANDARD)
cxwl_example(CxwlDepthThreadedCapture "cpp/ThreadedCapture/ThreadedCapture.cpp" STANDARD)
cxwl_example(CxwlDepthSelfTest "c/SelfTest/SelfTest.c" MINIMAL)
target_link_libraries(CxwlDepthSelfTest PRIVATE delayimp)
target_link_options(CxwlDepthSelfTest PRIVATE "/DELAYLOAD:CxwlDepthSdk.dll")
if(CXWL_BUILD_QT_VIEWER)
  add_subdirectory(QtViewer)
endif()
