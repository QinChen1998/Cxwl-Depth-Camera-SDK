cmake_minimum_required(VERSION 3.21)
project(CxwlDepthSdkBinary LANGUAGES C)
set(CXWL_RUNTIME_PROFILE STANDARD CACHE STRING "MINIMAL, STANDARD or FULL")
set_property(CACHE CXWL_RUNTIME_PROFILE PROPERTY STRINGS MINIMAL STANDARD FULL)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/CxwlDepthSdkConfig.cmake")
if(DEFINED CxwlDepthSdk_FOUND AND NOT CxwlDepthSdk_FOUND)
  message(FATAL_ERROR "${CxwlDepthSdk_NOT_FOUND_MESSAGE}")
endif()
# Export a public interface to the SDK DLL; the config defines its imported Runtime target.
add_library(CxwlDepthSdkExport INTERFACE)
set_target_properties(CxwlDepthSdkExport PROPERTIES EXPORT_NAME DepthSdk)
target_link_libraries(CxwlDepthSdkExport INTERFACE CXWL::Runtime)
target_include_directories(CxwlDepthSdkExport INTERFACE "$<INSTALL_INTERFACE:include>")
if(MSVC)
  target_compile_options(CxwlDepthSdkExport INTERFACE /utf-8)
endif()
install(TARGETS CxwlDepthSdkExport EXPORT CxwlDepthSdkTargets COMPONENT Development)
install(EXPORT CxwlDepthSdkTargets NAMESPACE CXWL::
  DESTINATION share/CxwlDepthSdk COMPONENT Development)
install(DIRECTORY include/ DESTINATION include COMPONENT Development)
install(FILES lib/CxwlDepthSdk.lib DESTINATION lib COMPONENT Development)
install(DIRECTORY cmake/ DESTINATION share/CxwlDepthSdk COMPONENT Development)
cxwl_install_runtime(PROFILE "${CXWL_RUNTIME_PROFILE}" COMPONENT Runtime)
