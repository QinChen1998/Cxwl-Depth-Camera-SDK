cmake_minimum_required(VERSION 3.21)
project(CxwlExampleLaunchers LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
foreach(name QuickStart Quality DeviceManagement PointCloud CppQuickStart ThreadedCapture SelfTest Viewer)
  set(target CxwlDepth${name})
  if(name STREQUAL Viewer)
    add_executable(${target} WIN32 Launcher.cpp)
    target_compile_definitions(${target} PRIVATE CXWL_GUI_LAUNCHER)
  else()
    add_executable(${target} Launcher.cpp)
  endif()
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(${target} PRIVATE "CXWL_SAMPLE_TARGET=L\"${target}.exe\"" NOMINMAX)
  target_compile_options(${target} PRIVATE /utf-8)
  target_link_libraries(${target} PRIVATE shell32)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${target}.rc"
    CONTENT "#define CXWL_FILE_DESCRIPTION \"CXWL SDK Example Launcher\"\n#define CXWL_BINARY_NAME \"${target}.exe\"\n#define CXWL_BINARY_TYPE VFT_APP\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/../common/ProductVersion.rc\"\n")
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}.rc")
endforeach()
