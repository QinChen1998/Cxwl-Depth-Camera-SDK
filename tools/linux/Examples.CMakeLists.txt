cmake_minimum_required(VERSION 3.22)
project(CxwlDepthExamples LANGUAGES C CXX)

find_package(CxwlDepthSdk 3.0 CONFIG REQUIRED
    PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../SDK/lib/cmake/CxwlDepthSdk"
    NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
find_package(Threads REQUIRED)

function(cxwl_example target source)
    add_executable(${target} ${source})
    target_include_directories(${target} PRIVATE common)
    target_compile_features(${target} PRIVATE c_std_11 cxx_std_17)
    target_link_libraries(${target} PRIVATE CXWL::DepthSdk ${ARGN})
    target_link_options(${target} PRIVATE "LINKER:-rpath-link,$<TARGET_FILE_DIR:CXWL::DepthSdk>/cxwl")
    set_target_properties(${target} PROPERTIES INSTALL_RPATH "$ORIGIN/../../SDK/lib")
    install(TARGETS ${target} RUNTIME DESTINATION bin)
endfunction()

cxwl_example(CxwlDepthQuickStart c/DepthSdkExample.c)
cxwl_example(CxwlDepthCppQuickStart cpp/CppQuickStart.cpp)
cxwl_example(CxwlDepthThreadedCapture cpp/ThreadedCapture.cpp Threads::Threads)
cxwl_example(CxwlDepthQuality c/DepthQuality.c)
cxwl_example(CxwlDepthPointCloud c/PointCloud.c m)
cxwl_example(CxwlDepthDeviceManagement c/DeviceManagement.c)

add_executable(CxwlDepthSelfTest c/SelfTest.c)
target_compile_features(CxwlDepthSelfTest PRIVATE c_std_11)
target_include_directories(CxwlDepthSelfTest PRIVATE common
    "$<TARGET_PROPERTY:CXWL::DepthSdk,INTERFACE_INCLUDE_DIRECTORIES>")
target_link_libraries(CxwlDepthSelfTest PRIVATE ${CMAKE_DL_LIBS} m)
set_target_properties(CxwlDepthSelfTest PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:CXWL::DepthSdk>"
    INSTALL_RPATH "$ORIGIN/../../SDK/lib")
install(TARGETS CxwlDepthSelfTest RUNTIME DESTINATION bin)
